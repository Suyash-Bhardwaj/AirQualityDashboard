import requests
import json

def get_live_data(country="IN", limit=10):
    """
    Fetches live air quality data from OpenAQ API for a given country.
    Default: India (IN)
    """
    url = f"https://api.openaq.org/v2/measurements?country_id={country}&limit={limit}"

    # Make the GET request
    response = requests.get(url)

    # Check if request succeeded
    if response.status_code == 200:
        data = response.json()
        print("✅ Data fetched successfully!")
        
        # Print the first few results (formatted)
        print(json.dumps(data["results"][:3], indent=4))
    else:
        print("❌ Failed to fetch data. Status code:", response.status_code)

# Test the function
if __name__ == "__main__":
    get_live_data()

requests.get(url) → sends an HTTP GET request to the API.

response.json() → converts the response into Python dictionary format.

data["results"] → contains the actual readings list.

json.dumps(..., indent=4) → pretty-prints the JSON so you can read it easily.

import pandas as pd
import matplotlib.pyplot as plt

def city_pollution_trend(df, city_name):
    """Plot pollution trend over time for a specific city."""
    
    # Filter data for the selected city
    city_data = df[df['city'].str.lower() == city_name.lower()].copy()

    if city_data.empty:
        print(f"⚠️ No data found for city: {city_name}")
        return

    # Sort by date/time
    city_data = city_data.sort_values(by='last_update')

    # Group by date and take mean pollutant_avg per day
    trend = (
        city_data.groupby(city_data['last_update'].dt.date)['pollutant_avg']
        .mean()
        .reset_index()
    )

    # Plot the trend
    plt.figure(figsize=(10, 5))
    plt.plot(trend['last_update'], trend['pollutant_avg'], marker='o', color='royalblue', linewidth=2)
    plt.title(f"Pollution Trend Over Time — {city_name}")
    plt.xlabel("Date")
    plt.ylabel("Average Pollutant Level")
    plt.grid(True)
    plt.tight_layout()

    # Save chart
    output_path = f"reports/summary_charts/{city_name}_trend.png"
    plt.savefig(output_path, dpi=300)
    plt.show()
    print(f"✅ Trend chart saved as: {output_path}")

import pandas as pd

def multi_city_trend(df, city_list):
    """
    Plot pollution trend over time for multiple cities on a single chart.
    """
    plt.figure(figsize=(10, 6))

    for city in city_list:
        # Filter data for each city
        city_data = df[df['city'].str.lower() == city.lower()].copy()

        if city_data.empty:
            print(f"⚠️ No data found for city: {city}")
            continue

        # Sort by time
        city_data = city_data.sort_values(by='last_update')

        # Group by date and average pollutant value
        trend = (
            city_data.groupby(city_data['last_update'].dt.date)['pollutant_avg']
            .mean()
            .reset_index()
        )

        # Plot each city’s trend line
        plt.plot(trend['last_update'], trend['pollutant_avg'], marker='o', linewidth=2, label=city)

    plt.title("City-wise Pollution Trend Over Time")
    plt.xlabel("Date")
    plt.ylabel("Average Pollutant Level")
    plt.legend(title="Cities")

    plt.tight_layout()

    # Save chart
    output_path = "reports/summary_charts/multi_city_trend.png"
